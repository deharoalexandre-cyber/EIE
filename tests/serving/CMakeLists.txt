cmake_minimum_required(VERSION 3.14)
project(eie_serving_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_executable(eie-serving-tests serving_unit.cpp)
target_include_directories(eie-serving-tests PRIVATE ../..)
target_link_libraries(eie-serving-tests PRIVATE Threads::Threads)
enable_testing()
add_test(NAME serving-functional COMMAND eie-serving-tests)

# Optional real HTTP routes with a deterministic fake backend (no llama/GPU).
set(EIE_HTTP_VENDOR "" CACHE PATH "llama.cpp vendor directory")
set(EIE_HTTP_LIBRARY "" CACHE FILEPATH "Compiled cpp-httplib library")
if(EIE_HTTP_VENDOR AND EIE_HTTP_LIBRARY)
    add_executable(eie-http-fixture http_fixture.cpp ../../server/api.cpp)
    target_include_directories(eie-http-fixture PRIVATE ../.. "${EIE_HTTP_VENDOR}" "${EIE_HTTP_VENDOR}/cpp-httplib")
    target_compile_definitions(eie-http-fixture PRIVATE HAS_HTTPLIB=1)
    target_link_libraries(eie-http-fixture PRIVATE "${EIE_HTTP_LIBRARY}" Threads::Threads)
endif()
